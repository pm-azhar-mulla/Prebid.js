PubMatic Hook Example

Demonstrates using hooks instead of RTD provider

Hook Implementation Example


// Register the hook
pbjs.que.push(function() {
    pbjs.requestBids.before(function(next, reqBidsConfigObj) {
        // Modify bid request configuration
        
        // Example: Add country data to ortb2
        const ortb2 = {
            user: {
                ext: {
                    ctr: 'US',
                }
            }
        };
        
        reqBidsConfigObj.ortb2Fragments = reqBidsConfigObj.ortb2Fragments || {};
        reqBidsConfigObj.ortb2Fragments.bidder = reqBidsConfigObj.ortb2Fragments.bidder || {};
        reqBidsConfigObj.ortb2Fragments.bidder.pubmatic = ortb2;
        
        // Continue with the auction
        return next(reqBidsConfigObj);
    }, 50);
});
